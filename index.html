function applyUnlockedData(data, email) { document.getElementById('stepEmailBlock').classList.add('hidden'); document.getElementById('stepOtpBlock').classList.add('hidden'); document.getElementById('authSuccessBlock').classList.remove('hidden'); document.getElementById('authenticatedEmailText').innerText = email; document.getElementById('authStatusMessage').innerHTML = ` Đã mở khóa bảo mật thành công!`; const accessedFolders = data.folders || []; const countEl = document.getElementById('lessonCountBadge'); const upgradeBox = document.getElementById('upgradeBox'); const upgradeStatusText = document.getElementById('upgradeStatusText'); const upgradeMissingBadge = document.getElementById('upgradeMissingBadge'); const retailPriceText = document.getElementById('retailPriceText'); const upgradeFeeTag = document.getElementById('upgradeFeeTag'); const savedAmountBadge = document.getElementById('savedAmountBadge'); if (accessedFolders.length > 0) { const accessedIds = accessedFolders.map(f => f.folderId); currentAllowedList = LESSONS_DATA.filter(item => { return accessedFolders.some(f => f.folderName.includes(item.code) || accessedIds.includes(item.folderId)); }); if (currentAllowedList.length === 0) { currentAllowedList = LESSONS_DATA.slice(0, accessedFolders.length); } countEl.innerText = `${currentAllowedList.length}/14 Đã mở`; renderLessons(currentAllowedList, false); // TÍNH TOÁN TRỰC TIẾP THEO ĐÚNG BẢNG CHÍNH SÁCH ĐỐI CHIẾU const totalMonths = 14; const purchased = currentAllowedList.length; const missing = Math.max(0, totalMonths - purchased); const retailTotal = missing * 500000; const paidAmount = purchased * 500000; let upgradeFee = 0; if (purchased >= 14) { upgradeFee = 0; } else if (purchased < 5) { upgradeFee = Math.max(0, 4900000 - paidAmount); } else if (purchased >= 5 && purchased <= 7) { upgradeFee = 2000000; } else if (purchased >= 8 && purchased <= 10) { upgradeFee = 1500000; } else if (purchased >= 11 && purchased <= 12) { upgradeFee = 1000000; } else if (purchased === 13) { upgradeFee = 500000; } const savedAmount = Math.max(0, retailTotal - upgradeFee); const savedPercent = retailTotal > 0 ? Math.round((savedAmount / retailTotal) * 100) : 0; // HIỂN THỊ SO SÁNH GIÁ NẾU CHƯA FULL if (missing > 0) { upgradeBox.classList.remove('hidden'); upgradeStatusText.innerText = `Đã sở hữu: ${purchased}/${totalMonths} Folder`; upgradeMissingBadge.innerText = `Còn thiếu: ${missing} Folder`; retailPriceText.innerText = `${retailTotal.toLocaleString('vi-VN')}đ (${missing} kỳ × 500k)`; upgradeFeeTag.innerText = `${upgradeFee.toLocaleString('vi-VN')}đ`; savedAmountBadge.innerHTML = ` Tiết kiệm: ${savedAmount.toLocaleString('vi-VN')}đ (${savedPercent}%)`; } else { upgradeBox.classList.add('hidden'); } } }
